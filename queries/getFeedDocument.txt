query ($id: String!) {
	getFeedDocument (id: $id) {
		__typename
		...on FeedDocument {
			id
			feed
			author
			title
			posted
			body
			body_html
			created
			updated
			url
			read
			html_len
		}
		...on FeedDoesNotExistError {
			message
		}
		...on InsufficientPerms {
			message
		}
	}
}
