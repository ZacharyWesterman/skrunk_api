mutation ($username: String!, $disabled: Boolean!) {
	updateUserDisabled (username: $username, disabled: $disabled) {
		__typename
		...on UserData {
			username
			display_name
			theme {
				colors {
					name
					value
				}
				sizes {
					name
					value
				}
			}
			perms
			last_login
			groups
			disabled_modules
			email
			failed_logins
			is_locked
			disabled
		}
		...on UserDoesNotExistError {
			message
		}
		...on InsufficientPerms {
			message
		}
	}
}
