mutation ($id: String!, $rfid: String) {
	relinkInventoryItem (id: $id, rfid: $rfid) {
		__typename
		...on Item {
			id
			created
			creator {
				username
				display_name
				last_login
				groups
				disabled
			}
			owner {
				username
				display_name
				last_login
				groups
				disabled
			}
			category
			type
			location
			blob {
				id
				created
				creator
				name
				ext
				mimetype
				size
				tags
				preview
				thumbnail
				hidden
				ephemeral
				references
				complete
			}
			description
			description_html
			rfid
		}
		...on InsufficientPerms {
			message
		}
		...on ItemDoesNotExistError {
			message
		}
		...on ItemExistsError {
			message
		}
	}
}
