mutation ($id: String!, $hidden: Boolean!) {
	setBlobHidden (id: $id, hidden: $hidden) {
		__typename
		...on Blob {
			id
			created
			creator
			name
			ext
			mimetype
			size
			tags
			previews
			thumbnail
			hidden
			ephemeral
			references
			complete
		}
		...on BlobDoesNotExistError {
			message
		}
		...on InsufficientPerms {
			message
		}
	}
}
