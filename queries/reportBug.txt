mutation ($text: String!) {
	reportBug (text: $text) {
		__typename
		...on BugReport {
			id
			created
			creator
			body
			body_html
			convo {
				created
				creator
				body
				body_html
			}
			resolved
		}
		...on BugReportCreationFailedError {
			message
		}
		...on InsufficientPerms {
			message
		}
	}
}
