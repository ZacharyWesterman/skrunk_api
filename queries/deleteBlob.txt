mutation ($id: String!) {
	deleteBlob (id: $id) {
		__typename
		...on Blob {
			id
			created
			creator
			name
			ext
			mimetype
			size
			tags
			previews
			thumbnail
			hidden
			ephemeral
			references
			complete
		}
		...on BlobDoesNotExistError {
			message
		}
		...on InsufficientPerms {
			message
		}
	}
}
