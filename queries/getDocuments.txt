query ($filter: DocumentSearchFilter!, $start: Int!, $count: Int!) {
	getDocuments (filter: $filter, start: $start, count: $count) {
		__typename
		...on DocumentList {
			documents {
				id
				title
				body
				body_html
				created
				creator {
					username
					display_name
					last_login
					groups
					disabled
				}
				updated
				updater {
					username
					display_name
					last_login
					groups
					disabled
				}
				blob_id
				blob_type
				tags
				shared_users {
					username
					display_name
					last_login
					groups
					disabled
				}
				shared_groups
			}
		}
		...on BadTagQuery {
			message
		}
	}
}
