mutation ($username: String!) {
	deleteUserTheme (username: $username) {
		__typename
		...on UserData {
			username
			display_name
			theme {
				colors {
					name
					value
				}
				sizes {
					name
					value
				}
			}
			perms
			last_login
			groups
			disabled_modules
			email
			failed_logins
			is_locked
			disabled
		}
		...on UserDoesNotExistError {
			message
		}
		...on InsufficientPerms {
			message
		}
	}
}
