mutation ($id: String!, $text: String!) {
	commentOnBug (id: $id, text: $text) {
		__typename
		...on BugReport {
			id
			created
			creator
			body
			body_html
			convo {
				created
				creator
				body
				body_html
			}
			resolved
		}
		...on BugReportDoesNotExistError {
			message
		}
		...on InsufficientPerms {
			message
		}
	}
}
