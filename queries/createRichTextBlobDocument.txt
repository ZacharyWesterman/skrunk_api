mutation ($title: String!) {
	createRichTextBlobDocument (title: $title) {
		__typename
		...on Document {
			id
			title
			body
			body_html
			created
			creator {
				username
				display_name
				last_login
				groups
				disabled
			}
			updated
			updater {
				username
				display_name
				last_login
				groups
				disabled
			}
			blob_id
			blob_type
			tags
			shared_users {
				username
				display_name
				last_login
				groups
				disabled
			}
			shared_groups
		}
		...on InsufficientPerms {
			message
		}
		...on BlobDocumentsNotSupported {
			message
		}
	}
}
